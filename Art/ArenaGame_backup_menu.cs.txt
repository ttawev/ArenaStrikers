        void StartMatch()
        {
            PlayerPrefs.SetInt("arena.hero", heroSel); PlayerPrefs.SetInt("arena.mode2", modeSel); PlayerPrefs.SetInt("arena.diff", diffSel);
            GUIUtility.keyboardControl = 0;
            match.T = tuning; match.PlayIntro = true; // выход команд (в тренировке его нет)
            match.Duration = Match.TimeOptions[timeSel]; match.WinGoals = Match.GoalOptions[goalSel];
            match.Setup(ModeList[modeSel], HeroList[heroSel], diffSel);
            view.Rebuild(match);
            input.ClearPending(); acc = 0;
            goalReplay.End(match); goalReplay.Clear(); replayIn = 0; Time.timeScale = 1;
            ui.Show(MenuUI.Screen.None); showcase.Hide(); view.ShowYouMark = true;
            state = State.Playing;
        }

        /// <summary>В меню: на фоне играют боты (без людей), экран меню — указанный.</summary>
        void GoToMenu(MenuUI.Screen screen)
        {
            Time.timeScale = 1; input.StopRumble(); netMenu = false;
            match.T = tuning; match.LocalSlot = 0; match.PlayIntro = false; // фоновый матч в меню — без выхода команд
            match.Duration = Field.MATCH_TIME; match.WinGoals = (int)Field.WIN_GOALS;
            match.Setup(GameMode.FourVsFour, HeroList[heroSel], 1);
            foreach (var p in match.Players) p.HumanSlot = -1; // все — боты
            view.Rebuild(match); view.ShowYouMark = false;
            goalReplay.End(match); goalReplay.Clear(); replayIn = 0;
            attractAcc = 0; state = State.Menu;
            ui.Show(screen);
        }

        /// <summary>Фоновый матч ботов за меню (без вибрации и тряски камеры).</summary>
        void AttractStep(float dt)
        {
            attractAcc += dt;
            int n = 0;
            while (attractAcc >= DT && n++ < 12)
            {
                match.Step(DT, noInput); attractAcc -= DT;
                foreach (var e in match.Events)
                {
                    if (e.Kind == FxKind.Burst) view.Burst(e.X, e.Y, Mat.Of(e.Color), e.Count, e.Amount);
                    else if (e.Kind == FxKind.Goal) view.Burst(e.X, e.Y, e.Team == Team.Home ? Mat.Home : Mat.Away, 60, 1.6f);
                    else if (e.Kind == FxKind.Impact) view.Impact(e.X, e.Y, e.Amount);
                }
                match.Events.Clear();
                if (match.Phase == MatchPhase.Ended) match.NewMatch();
            }
            if (attractAcc > DT) attractAcc = 0;
        }

        void Update()
        {
            // при записи Unity Recorder с постоянной частотой кадров время идёт по кадрам записи, а не по часам
            float dt = Mathf.Min(Time.captureDeltaTime > 0 ? Time.captureDeltaTime : Time.unscaledDeltaTime, 0.05f);
            uiTime += dt;
            TrackShowState();
            ControlsTick(dt);
            MenuInput();
            if (state == State.Playing || state == State.Replay) ClipKeys();
            if (net != null) NetUpdate(dt);
            if (state == State.Playing && IsClient)
            {   // друг: матч считает хост, мы шлём ввод и показываем его снимки
                input.Poll(match.Local.Hero);
                var c = input.Take(match.Local.Hero); lastCmd = c;
                net.ClientTick(match, net.InterpDelay, e => Fx(e, false, true)); // все — по снимкам хоста (чуть в прошлом)
                net.ClientStep(match, netMenu ? default : c, dt);                 // свой герой — сразу, по предсказанию; ввод — хосту
                ClientGoalCheck();
                if (match.Phase == MatchPhase.Ended) { state = State.Ended; input.StopRumble(); }
            }
            else if (state == State.Ended && IsClient) net.ClientTick(match, net.InterpDelay, e => Fx(e, false, true));
            else if (state == State.Playing)
            {
                input.Poll(match.Local.Hero);
                bool slowNow = net == null && slowHold > 0; // включают гол (slowMo) и кинокамера (cine.slowKicks)
                slowHold -= dt;
                slow = Mathf.MoveTowards(slow, slowNow ? slowAmt : 1, dt * (slowNow ? 10 : 2.2f));
                if (hitstop > 0) hitstop -= dt; // freeze-frame on strong hits: the sim waits, input is kept
                else acc += dt * slow;
                while (acc >= DT)
                {
                    cmds[0] = input.Take(match.Local.Hero); lastCmd = cmds[0];
                    if (netMenu) cmds[0] = default;
                    if (IsHostOnline) for (int k = 1; k < cmds.Length; k++) cmds[k] = net.TakeInput(k);
                    bool frozen = match.Freeze > 0;
                    match.Step(DT, cmds);
                    simTime += DT; acc -= DT;
                    if (!frozen && match.Intro <= 0 && replay && net == null && match.Mode != GameMode.Training) goalReplay.Record(match, simTime, replaySeconds + 0.5f);
                    if (net != null) net.AddEvents(match.Events);
                    HandleEvents();
                    if (match.Phase == MatchPhase.Ended) { state = State.Ended; input.StopRumble(); break; }
                }
                if (replayIn > 0 && state == State.Playing) { replayIn -= dt; if (replayIn <= 0) StartReplay(); }
            }
            else if (state == State.Replay)
            {
                if (!goalReplay.Tick(match, dt * replaySpeed, e => Fx(e, false))) EndReplay();
                match.Events.Clear();
            }
            else if (state == State.Reel) ReelStep(dt);
            else if (state == State.Menu || state == State.Lobby) AttractStep(dt);
            else match.Events.Clear();
            if (state == State.Menu || state == State.Lobby)
            {
                ui.Tick(dt);
                if (ui.Showcase) { showcase.Show(HeroList[heroSel], state == State.Lobby ? netTeam : Team.Home); showcase.Tick(dt, cam.transform.position); }
                else showcase.Hide();
            }
            if (IsHostOnline && (state == State.Playing || state == State.Ended)) net.HostTick(match, simTime, dt);
            foreach (var line in match.Log) Debug.LogWarning("[Arena] " + line); // автоголы — в Logs/arena-runtime.log
            match.Log.Clear();
            input.UpdateRumble(dt);
            denyFlash = Mathf.Max(0, denyFlash - dt);
            HeroModel.Tune = heroArt;
            view.SetHeroLight(heroArt.ambientSide, heroArt.ambientGround, heroArt.fillLight);
            MvpLightTick(); // экран MVP: свой свет на лучшем игроке (Game/ArenaGame.MvpLight.cs)
            YouMarker.Cfg = youMarker;
            GoalFx.NetPunch = netPunch; GoalFx.NetStiff = netStiff; GoalFx.NetDamp = netDamp; GoalFx.ConfettiCount = confetti;
            StadiumView.GoalFlash = goalFlash; StadiumView.GoalFlashTime = goalFlashTime;
            view.SetBloom(heroArt.bloom);
            if (state != State.Playing) { slow = 1; slowHold = 0; if (state == State.Reel) Time.timeScale = ReelPlaying && reelGap <= 0 ? reelSpeed : 1; else if (state != State.Replay) Time.timeScale = 1; }
            else Time.timeScale = slow; // частицы тоже замедляются
            float vdt = state == State.Replay ? dt * replaySpeed : state == State.Reel ? (ReelPlaying ? dt * reelSpeed : 0) : dt * slow;
            HeroModel.FrameDt = state == State.Paused || hitstop > 0 ? 0 : vdt;
            view.Sync(match, Time.time, vdt, state == State.Playing ? lastCmd : default);
            toastT = Mathf.Max(0, toastT - dt);
            UpdateCamera(dt);
            if (state == State.Reel) TrackFrame();
            bool cineFx = cine.postFx && (CineOn || ReelPlaying || showFx);
            cineFxW = Mathf.MoveTowards(cineFxW, cineFx ? 1 : 0, dt * 3);
            view.SetCineFx(cineFxW, showFx ? showFocus : Vector3.Distance(cam.transform.position, cineCam.Focus), cine);
            Online.SnapRate = snapsPerSecond; Online.MinDelay = minBuffer; Online.PredictOn = predictOwnHero; Online.MaxQueue = inputQueue; Online.SimLagMs = simulatedPing;
            sound.Master = masterVolume; sound.Sfx = sfxVolume; sound.Crowd = crowdVolume; sound.Music = musicVolume;
            sound.Sync(match, cam, dt, state == State.Playing || state == State.Paused || state == State.Ended || state == State.Replay || ReelPlaying, state == State.Replay || state == State.Reel);
        }

        void HandleEvents()
        {
            foreach (var e in match.Events) Fx(e, true);
            match.Events.Clear();
        }

        /// <summary>Эффекты события. live = false — событие из повтора гола: без вибрации, заморозки кадра и нового повтора.</summary>
        void Fx(SimEvent e, bool live, bool remote = false)
        {
                // друг: гол мог прийти и событием, и по изменению счёта (ClientGoalCheck) — показываем один раз
                if (e.Kind == FxKind.Goal && remote && uiTime - goalT0 < 1.5f) return;
                if (state != State.Menu && state != State.Lobby) sound.OnEvent(match, e, live, remote); // фоновый матч в меню — без звуков ударов
                switch (e.Kind)
                {
                    case FxKind.Burst: view.Burst(e.X, e.Y, Mat.Of(e.Color), e.Count, e.Amount); break;
                    case FxKind.Shake: shake = Mathf.Max(shake, e.Amount); break;
                    case FxKind.Goal:
                        view.Burst(e.X, e.Y, e.Team == Team.Home ? Mat.Home : Mat.Away, 60, 1.6f);
                        view.GoalHit(match, e.Team, e.X, e.Y, true);
                        if (live || remote)
                        {
                            goalT0 = uiTime; goalTeam = e.Team; goalScoreH = match.ScoreHome; goalScoreA = match.ScoreAway;
                            goalScorer = FindScorer(e.Team);
                            if (live && net == null && slowMo && match.Mode != GameMode.Training) { slowHold = slowMoAfter; slow = slowMoAmount; slowAmt = slowMoAmount; }
                        }
                        if (live && replay && net == null && match.Mode != GameMode.Training && goalReplay.HasFrames) replayIn = Mathf.Max(replayDelay, celebrate);
                        break;
                    case FxKind.Rumble: if ((live || remote) && e.Player == match.Local) input.Rumble(e.Amount); break;
                    case FxKind.Denied: if ((live || remote) && e.Player == match.Local) denyFlash = 0.25f; break;
                    case FxKind.Impact:
                        view.Impact(e.X, e.Y, e.Amount);
                        if (live && net == null && CineOn && cine.slowKicks && e.Amount >= cine.slowMinPower && match.Freeze <= 0)
                        { slowHold = Mathf.Max(slowHold, cine.slowTime); slowAmt = cine.slowAmount; } // кинокамера: удар замедленно
                        if (live && e.Amount >= 0.5f) { hitstop = Mathf.Max(hitstop, (0.045f + 0.075f * Mathf.Min(e.Amount, 1.4f)) * hitstopScale); acc = 0; }
                        shake = Mathf.Max(shake, e.Amount * 10 * shakeScale);
                        fovPunch = Mathf.Max(fovPunch, Mathf.Clamp01(e.Amount) * fovKick);
                        view.Burst(e.X, e.Y, Mat.Mist, (int)(6 + 10 * e.Amount), 0.6f + 0.6f * e.Amount);
                        break;
                    case FxKind.Receive: break; // звук — в MatchAudio.OnEvent
                }
        }

        // ---------- повтор гола и съёмка ----------
        void StartReplay()
        {
            if (!goalReplay.Begin(match, replaySeconds, 0.6f)) return;
            state = State.Replay; Time.timeScale = replaySpeed; // частицы и анимации стадиона тоже замедляются
            input.StopRumble(); view.ClearTrails();
        }

        void EndReplay()
        {
            goalReplay.End(match);
            Time.timeScale = 1; state = State.Playing;
            input.ClearPending(); acc = 0; view.ClearTrails();
        }

        // ---------- хайлайты ботов ----------
        void StartReel()
        {
            baseMatch = match;
            scan = new HighlightScan(tuning, highlights);
            reelReady = false; reelIdx = -1; reelGap = 0; reelShot = CineCam.Shot.Broadcast;
            ui.Show(MenuUI.Screen.None); showcase.Hide(); view.ShowYouMark = false;
            cineCam.Seed(cam.transform.position, cam.transform.rotation);
            state = State.Reel; Time.timeScale = 1;
        }

        void ReelStep(float dt)
        {
            match.Events.Clear();
            if (!scan.Finished)
            {
                scan.Work(25); // ~25 мс за кадр — игра не зависает, пока ищем моменты
                if (scan.Finished)
                {
                    if (scan.Best.Count == 0) { ExitReel(); Toast("Голов не нашлось — попробуй ещё раз"); return; }
                    reelReady = true; WriteReelLog();
                }
                return;
            }
            if (!ReelPlaying) return;
            if (reelGap > 0) { reelGap -= dt; if (reelGap <= 0) NextAngle(); return; }
            var hl = scan.Best[reelIdx];
            if (!hl.Rec.Tick(match, dt * reelSpeed, e => Fx(e, false))) NextAngle();
            match.Events.Clear();
        }

        /// <summary>Следующий ракурс того же момента: трансляция → за мячом замедленно → облёт забившего.</summary>
        void NextAngle()
        {
            var hl = scan.Best[reelIdx]; var h = highlights; float g = hl.GoalT;
            reelAngle++;
            if (reelAngle == 0) { ReelShot(hl, CineCam.Shot.Broadcast, g - h.leadSeconds, g + 0.9f, 1, 0.1f); return; }
            if (reelAngle == 1) { ReelShot(hl, CineCam.Shot.Chase, g - h.slowSeconds, g + 0.6f, h.slowSpeed, 0.15f); return; }
            if (reelAngle == 2 && h.orbit && hl.Scorer != null) { ReelShot(hl, CineCam.Shot.Orbit, g + 0.5f, g + 0.5f + h.orbitSeconds, 1, 0); return; } // облёт — когда мяч уже в сетке
            NextHighlight();
        }

        void ReelShot(Highlight hl, CineCam.Shot shot, float from, float to, float speed, float hold)
        {
            reelShot = shot; reelSpeed = speed;
            cineCam.Cut(); view.ClearTrails();
            if (!hl.Rec.BeginRange(match, from, to, hold)) { NextAngle(); return; }
            if (shot == CineCam.Shot.Orbit)
            {   // плашка гола поверх облёта: счёт уже с этим голом
                goalT0 = uiTime; goalTeam = match.BannerTeam; goalScoreH = match.ScoreHome; goalScoreA = match.ScoreAway;
                goalScorer = hl.Scorer != null ? match.Players.IndexOf(hl.Scorer) : -1;
            }
        }

        void NextHighlight()
        {
            reelIdx++; reelAngle = -1;
            if (reelIdx >= scan.Best.Count) { ExitReel(); Toast("Хайлайты закончились — останови запись"); return; }
            match = scan.Best[reelIdx].M;
            match.Phase = MatchPhase.Playing; match.LocalSlot = 0;
            view.Rebuild(match); view.ShowYouMark = false;
            if (reelIdx > 0 && highlights.gap > 0) reelGap = highlights.gap; else NextAngle();
        }

        void ExitReel()
        {
            if (scan != null && reelIdx >= 0 && reelIdx < scan.Best.Count) scan.Best[reelIdx].Rec.End(match);
            SaveTrack();
            if (baseMatch != null) match = baseMatch;
            scan = null; baseMatch = null; reelIdx = -1; reelGap = 0; reelSpeed = 1;
            GoToMenu(MenuUI.Screen.Main);
        }

        /// <summary>Записать положение мяча на экране в этом кадре (вызывается после камеры).</summary>
        void TrackFrame()
        {
            if (track == null || !ReelPlaying) return;
            var b = match.Ball;
            Vector3 vp = cam.WorldToViewportPoint(W.Pos(b.X, b.Y, b.Z));
            track.Append(trackFrame++).Append(';').Append(reelIdx + 1).Append(';').Append(reelAngle).Append(';').Append(reelGap > 0 ? 1 : 0).Append(';')
                 .Append(vp.x.ToString("0.0000", System.Globalization.CultureInfo.InvariantCulture)).Append(';')
                 .Append(vp.y.ToString("0.0000", System.Globalization.CultureInfo.InvariantCulture)).Append('\n');
        }

        void SaveTrack()
        {
            if (track == null) return;
            try
            {
                string dir = System.IO.Path.Combine(Application.dataPath, "..", "Recordings");
                System.IO.Directory.CreateDirectory(dir);
                System.IO.File.WriteAllText(System.IO.Path.Combine(dir, "track_" + System.DateTime.Now.ToString("yyyyMMdd_HHmm") + ".csv"), track.ToString());
            }
            catch (System.Exception ex) { Debug.LogWarning("[Arena] track: " + ex.Message); }
            track = null;
        }

        /// <summary>Список моментов в Recordings/highlights.txt — по нему Claude монтирует запись.</summary>

        {
            try
            {
                string dir = System.IO.Path.Combine(Application.dataPath, "..", "Recordings");
                System.IO.Directory.CreateDirectory(dir);
                var sb = new System.Text.StringBuilder();
                sb.AppendLine("Хайлайты ботов " + System.DateTime.Now.ToString("yyyy-MM-dd HH:mm") + ": матчей " + scan.Total + ", голов " + scan.Goals

                              + "; между моментами чёрный кадр " + highlights.gap + " с");

                    sb.AppendLine((i + 1) + ". " + scan.Best[i].Desc + "  (баллы " + scan.Best[i].Score.ToString("0.0") + ", матч " + scan.Best[i].MatchNo + ")");

            if (camMode != prev)

        // ---------- menus: keyboard, gamepad, mouse ----------
        void MenuInput()
        {
            var gp = Gamepad.current; var kb = Keyboard.current;
            bool typing = (state == State.Menu || state == State.Lobby) && GUIUtility.keyboardControl != 0; // печатают имя или код комнаты — клавиши не трогаем
            if (typing) kb = null;
            bool confirm = (gp != null && gp.buttonSouth.wasPressedThisFrame) || (kb != null && kb.enterKey.wasPressedThisFrame);
            bool back = (gp != null && gp.buttonEast.wasPressedThisFrame) || (kb != null && kb.backspaceKey.wasPressedThisFrame);
            bool pause = (gp != null && gp.startButton.wasPressedThisFrame) || (kb != null && (kb.escapeKey.wasPressedThisFrame || kb.pKey.wasPressedThisFrame));
            switch (state)
            {
                case State.Menu: // меню на UI Toolkit само читает геймпад и клавиатуру
                    if (kb != null && kb.f8Key.wasPressedThisFrame && net == null) StartReel();
                    break;
                case State.Reel:
                    if (reelReady && confirm) { reelReady = false; reelIdx = -1; track = new System.Text.StringBuilder("frame;highlight;angle;gap;bx;by\n"); trackFrame = 0; NextHighlight(); }
                    else if (back || pause) ExitReel();
                    break;
                case State.Playing:
                    if (net != null) { if (pause) netMenu = !netMenu; else if (netMenu && back) ExitOnline(); } // онлайн матч не ставится на паузу
                    else if (match.Intro > 0 && (confirm || pause)) { match.SkipIntro(); input.ClearPending(); } // пропустить выход команд
                    else if (pause) { state = State.Paused; input.StopRumble(); }
                    break;
                case State.Lobby: break;
                case State.Paused:
                    if (pause || confirm) { state = State.Playing; input.ClearPending(); acc = 0; }
                    else if (back) GoToMenu(MenuUI.Screen.Main);
                    break;
                case State.Replay:
                    if (confirm || pause || back) EndReplay(); // пропустить повтор
                    break;
                case State.Ended:
                    if (net != null)
                    {   // онлайн: обратно в комнату — там можно сменить героя и команду и начать следующий матч
                        if (confirm) ReturnToRoom();
                        else if (back) ExitOnline();
                    }
                    else if (confirm) StartMatch();
                    else if (back) GoToMenu(MenuUI.Screen.Main);
                    break;
            }
        }

        /// <summary>Кто забил: последний коснувшийся из забившей команды, иначе ближайший к мячу из неё.</summary>
        int FindScorer(Team team)
        {
            var b = match.Ball;
            SimPlayer best = b.Last != null && b.Last.Team == team ? b.Last : b.Kicker != null && b.Kicker.Team == team ? b.Kicker : null;
            if (best == null)
            {
                float bd = float.MaxValue;
                foreach (var p in match.Players)
                    if (p.Team == team && !p.Dummy) { float d = (p.X - b.X) * (p.X - b.X) + (p.Y - b.Y) * (p.Y - b.Y); if (d < bd) { bd = d; best = p; } }
            }
            return best != null ? match.Players.IndexOf(best) : -1;
        }

        bool GoalCamOn => goalCam && state == State.Playing && match.Freeze > 0 && match.Banner != "" && match.Mode != GameMode.Training
                          && goalScorer >= 0 && goalScorer < match.Players.Count;

        Vector3 TargetCamPos(bool snap)
        {
            var you = match.Local; var b = match.Ball;
            if (GoalCamOn)
            {   // гол: камера подлетает к забившему
                var sp = match.Players[goalScorer];
                Vector3 f = W.Pos(sp.X, sp.Y);
                float hl = Field.FH / 2 * W.S, hw = Field.FW / 2 * W.S;
                f.x = Mathf.Clamp(f.x, -hl + 1.5f, hl - 1.5f); f.z = Mathf.Clamp(f.z, -hw + 1, hw - 1);
                return f - Quaternion.Euler(snap ? goalCamTilt : camTilt, 0, 0) * Vector3.forward * goalCamDistance;
            }
            if (ShowCam(out var sf, out var sd, out var stl)) return sf - Quaternion.Euler(snap ? stl : camTilt, 0, 0) * Vector3.forward * sd; // выход команд, MVP
            float wy = state == State.Replay ? 0.1f : state == State.Menu || state == State.Lobby ? 0.2f : clipCam ? 0.45f : 0.6f; // повтор и съёмка — больше на мяч
            float tx = you.X * wy + b.X * (1 - wy), ty = you.Y * wy + b.Y * (1 - wy);
            Vector3 focus = W.Pos(tx, ty);
            // keep the view over the pitch
            float halfLen = Field.FH / 2 * W.S, halfWid = Field.FW / 2 * W.S;
            focus.x = Mathf.Clamp(focus.x, -halfLen + 5, halfLen - 5);
            focus.z = Mathf.Clamp(focus.z, -halfWid + 3, halfWid - 3);
            float dist = state == State.Replay ? replayDistance : clipCam ? clipDistance : cameraDistance;
            var rot = Quaternion.Euler(snap ? CamTiltTarget : camTilt, 0, 0);
            return focus - rot * Vector3.forward * dist;
        }

        float CamTiltTarget => GoalCamOn ? goalCamTilt : ShowCam(out _, out _, out var st) ? st : state == State.Replay ? replayTilt : clipCam ? clipTilt : cameraTilt;

        void UpdateCamera(float dt)
        {
            if ((state == State.Menu || state == State.Lobby) && ui.Showcase)
            {   // экран выбора героя: камера у подиума
                HeroShowcase.CameraPose(out var sp, out var sr);
                float k = 1 - Mathf.Exp(-4 * dt);
                camPos = Vector3.Lerp(camPos, sp, k); camRot = Quaternion.Slerp(camRot, sr, k);
                cam.fieldOfView = Mathf.Lerp(cam.fieldOfView, 40, k);
                cam.transform.SetPositionAndRotation(camPos, camRot);
                return;
            }
            if (ShowPose(dt, out var spp, out var spr, out var sfov))
            {   // выход команд и MVP: свои киношные планы (Game/ArenaGame.Show.cs)
                cam.fieldOfView = sfov;
                cam.transform.SetPositionAndRotation(spp, spr);
                camPos = spp; camRot = spr; float ex = spr.eulerAngles.x; camTilt = ex > 180 ? ex - 360 : ex; // дальше обычная камера плавно уедет отсюда
                shake = 0;
                return;
            }
            if (CineOn || state == State.Reel)
            {   // кинокамера для клипов: Game/CineCam.cs; в хайлайтах план задаёт режиссёр хайлайтов
                bool reel = state == State.Reel;
                CineCam.Shot? shot = reel ? reelShot : camMode == 2 ? (CineCam.Shot?)null : (CineCam.Shot)(camMode - 3);
                var hero = reel ? (ReelPlaying ? scan.Best[reelIdx].Scorer : null) : GoalCamOn ? match.Players[goalScorer] : null;
                cineCam.Tick(match, cine, shot, hero, state == State.Replay || reel, state == State.Paused ? 0 : dt, out var cp, out var cr);
                fovPunch = Mathf.Max(0, fovPunch - dt * 18);
                cam.fieldOfView = cine.fov - fovPunch * 0.6f;
                Vector3 jit = shake > 0 ? Random.insideUnitSphere * shake * 2f * W.S : Vector3.zero;
                cam.transform.SetPositionAndRotation(cp + jit, cr);
                camPos = cp; camRot = cr; // выключишь — обычная камера плавно уедет отсюда
                shake = Mathf.Max(0, shake - dt * 30);
                return;
            }
            camTilt = Mathf.Lerp(camTilt, CamTiltTarget, 1 - Mathf.Exp(-4 * dt));
            var target = TargetCamPos(false);
            camPos = Vector3.Lerp(camPos, target, 1 - Mathf.Exp((GoalCamOn ? -2.6f : ShowCamOn ? -2.2f : state == State.Replay ? -3 : -5) * dt));
            fovPunch = Mathf.Max(0, fovPunch - dt * 18);
            cam.fieldOfView = cameraFov - fovPunch;
            camRot = Quaternion.Slerp(camRot, Quaternion.Euler(camTilt, 0, 0), 1 - Mathf.Exp(-6 * dt));
            if (ShowCamOn) camTilt = Mathf.Lerp(camTilt, CamTiltTarget, 1 - Mathf.Exp(-2 * dt)); // наклон доводим мягче
            cam.transform.rotation = camRot;
            Vector3 jitter = shake > 0 ? new Vector3(Random.value - 0.5f, 0, Random.value - 0.5f) * shake * 4f * W.S : Vector3.zero;
            cam.transform.position = camPos + jitter;
            shake = Mathf.Max(0, shake - dt * 30);
        }

        GUIStyle big, mid, small, banner, huge;

        /// <summary>Пружинка 0 → 1 с перелётом (для надписи и счёта).</summary>
        static float Spring(float t) => t <= 0 ? 0 : 1 - Mathf.Exp(-7 * t) * Mathf.Cos(13 * t);

        /// <summary>«ГОЛ!» и крупный счёт: выскакивают пружинкой, цифра забившей команды меняется с «прыжком».</summary>
        void GoalBannerPlain(float w, float h) // прежняя простая надпись (новая — ArenaGame.Banner.cs)
        {
            float t = uiTime - goalT0;
            Color col = match.BannerTeam == Team.Home ? Mat.Home : Mat.Away;
            var r = new Rect(0, h * 0.24f, w, 100);
            float s = t > 6 ? 1 : Spring(t);
            GUIUtility.ScaleAroundPivot(Vector2.one * Mathf.Max(0.01f, s), r.center);
            Shadowed(r, BannerText(), banner, col);
            GUI.matrix = Matrix4x4.identity;
            if (match.Mode == GameMode.Training || t > 30) return;
            float s2 = Spring(t - 0.15f);
            if (s2 <= 0) return;
            float y = h * 0.24f + 100;
            var row = new Rect(w / 2 - 230, y, 460, 120);
            GUIUtility.ScaleAroundPivot(Vector2.one * Mathf.Max(0.01f, s2), row.center);
            bool flip = t > 0.6f; // сначала старый счёт, потом цифра забивших «прыгает» на новую
            int hs = goalScoreH - (goalTeam == Team.Home && !flip ? 1 : 0), aw = goalScoreA - (goalTeam == Team.Away && !flip ? 1 : 0);
            Shadowed(new Rect(w / 2 - 40, y - 6, 80, 120), ":", huge, Color.white);
            Digit(hs, new Rect(w / 2 - 200, y, 160, 120), Mat.Home, goalTeam == Team.Home && flip ? t - 0.6f : -1);
            Digit(aw, new Rect(w / 2 + 40, y, 160, 120), Mat.Away, goalTeam == Team.Away && flip ? t - 0.6f : -1);
            GUI.matrix = Matrix4x4.identity;
        }

        void Digit(int v, Rect r, Color c, float popT)
        {
            var old = GUI.matrix;
            if (popT >= 0) { float k = 1 + 0.7f * Mathf.Exp(-6 * popT) * Mathf.Cos(11 * popT); GUIUtility.ScaleAroundPivot(Vector2.one * k, r.center); }
            Shadowed(r, v.ToString(), huge, c);
            GUI.matrix = old;
        }

        static Texture2D disc, ringTex;
        static Texture2D MakeCircle(bool ring)
        {
            const int N = 128; var t = new Texture2D(N, N, TextureFormat.RGBA32, false); var px = new Color32[N * N];
            for (int y = 0; y < N; y++) for (int x = 0; x < N; x++)
                {
                    float d = Vector2.Distance(new Vector2(x + 0.5f, y + 0.5f), new Vector2(N / 2f, N / 2f));
                    float a = Mathf.Clamp01(N / 2f - d);
                    if (ring) a *= Mathf.Clamp01(d - (N / 2f - 7));
                    px[y * N + x] = new Color32(255, 255, 255, (byte)(a * 255));
                }
            t.SetPixels32(px); t.Apply(); t.hideFlags = HideFlags.HideAndDontSave; return t;
        }

        void JoinServer()
        {
            PlayerPrefs.SetInt("arena.hero", heroSel);
            netError = ""; netTeam = Team.Away;

            net.JoinServer(ServerAddr, netName, HeroList[heroSel]);

        void ExitOnline()
        {
            if (net != null)
            {
                net.MatchStarted -= OnNetStart;
                if (net.State != Online.Phase.Failed) net.Leave();
                net = null;
            }
            GoToMenu(MenuUI.Screen.Online);
        }

        void NetUpdate(float dt)
        {
            if (net.State == Online.Phase.Failed) { netError = net.Status; ExitOnline(); return; }
            net.Tick(dt);
            if (net.ViaServer)
            {   // «Найти матч»: команду и героя (если занят в команде) назначает сервер
                netTeam = net.MyTeam;
                int hi = System.Array.IndexOf(HeroList, net.MyHero);
                if (hi >= 0) heroSel = hi;
            }
            if (net.IsHost)
                while (net.LeftSlots.Count > 0)
                {
                    int slot = net.LeftSlots.Dequeue();
                    var p = match.Players.Find(x => x.HumanSlot == slot && slot != 0);
                    if (p != null && state != State.Lobby) { p.HumanSlot = -1; Toast("Игрок вышел — за него играет бот"); } // дальше героем управляет бот
                }
        }

        /// <summary>Хост: начать матч со всеми в комнате (свободные места — боты).</summary>
        void StartOnlineMatch()
        {
            var humans = new System.Collections.Generic.List<Match.Seat>();
            int home = 0, away = 0;
            foreach (var st in net.Roster) { humans.Add(new Match.Seat { Team = st.Team, Hero = st.Hero, Slot = st.Slot }); if (st.Team == Team.Home) home++; else away++; }
            var mode = net.Mode;
            int need = Mathf.Max(home, away);
            if (Match.TeamSize(mode) < need) mode = Match.ModeFor(need);
            match.T = tuning; match.LocalSlot = 0; match.PlayIntro = true;
            match.Duration = net.MatchTime; match.WinGoals = net.WinGoals;
            match.SetupLineup(mode, humans, net.Difficulty);
            view.Rebuild(match);
            input.ClearPending(); acc = 0; hitstop = 0; goalReplay.Clear(); Time.timeScale = 1; netMenu = false;
            net.SendStart(match);
            ui.Show(MenuUI.Screen.None); showcase.Hide(); view.ShowYouMark = true;
            state = State.Playing;
        }

        /// <summary>Друг: хост начал матч.</summary>
        /// <summary>Друг: снимки идут без гарантии доставки, событие гола могло потеряться — тогда гол замечаем по счёту.</summary>
        void ClientGoalCheck()
        {
            int h = match.ScoreHome, a = match.ScoreAway;
            if (h + a > netScoreH + netScoreA && match.Mode != GameMode.Training)
            {
                var b = match.Ball;
                Fx(new SimEvent { Kind = FxKind.Goal, Team = h > netScoreH ? Team.Home : Team.Away, X = b.X, Y = Mathf.Clamp(b.Y, -20, Field.FH + 20) }, false, true);
            }
            netScoreH = h; netScoreA = a;
        }
        int netScoreH, netScoreA;

        static string PingText(int ms) => ms < 0 ? "пинг —" : "пинг " + ms + " мс";

        /// <summary>Строка в углу: у друга — свой пинг до хоста, у хоста — пинг каждого друга.</summary>

            if (!net.IsHost) return (net.ViaServer ? "сервер · " : "") + PingText(net.MyPing) + (simulatedPing > 0 ? " + имитация " + simulatedPing + " мс" : "") + "   буфер " + Mathf.RoundToInt(net.InterpDelay * 1000) + " мс" + (predictOwnHero ? "   предсказание" : "");

        }

        static bool Toggle(Rect r, bool on, string text)
        {
            var old = GUI.backgroundColor;
            GUI.backgroundColor = on ? Mat.Accent : new Color(0.3f, 0.4f, 0.4f);
            bool clicked = GUI.Button(r, text, new GUIStyle(GUI.skin.button) { wordWrap = true, fontSize = 13 });
            GUI.backgroundColor = old;
            return clicked;
        }

    /// <summary>Editor play sessions: errors and warnings go to Logs/arena-runtime.log so they can be read without the Console.</summary>
    static class RuntimeLog
    {
        static string path;
        public static void Start()
        {
#if UNITY_EDITOR
            path = System.IO.Path.Combine(System.IO.Directory.GetCurrentDirectory(), "Logs", "arena-runtime.log");
            try { System.IO.File.WriteAllText(path, "play started " + System.DateTime.Now.ToString("HH:mm:ss") + "\n"); } catch { }
            Application.logMessageReceived -= OnLog;
            Application.logMessageReceived += OnLog;
#endif
        }
        static void OnLog(string msg, string stack, LogType type)
        {
            if (type == LogType.Log || path == null) return;
            try { System.IO.File.AppendAllText(path, type + ": " + msg + "\n" + (type == LogType.Exception || type == LogType.Error ? stack + "\n" : "")); } catch { }
        }
    }
